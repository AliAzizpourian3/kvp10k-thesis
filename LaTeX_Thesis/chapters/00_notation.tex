\chapter*{List of Symbols}
\addcontentsline{toc}{chapter}{List of Symbols}

\begingroup
\small
\renewcommand{\arraystretch}{1.12}
\begin{longtable}{@{}p{0.27\textwidth}p{0.67\textwidth}@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} \\ \midrule
$n_{\text{boxes}}, A_{\text{total}}, \mu_A, \sigma_A$
  & Number of annotation boxes, their total area, mean area, and area standard deviation. \\
$\mu_w, \mu_h, \mu_r, \mu_x, \mu_y$
  & Mean box width, height, aspect ratio, horizontal centre, and vertical centre. \\
$s_v, s_h, d_{\text{spacing}}$
  & Vertical spread, horizontal spread, and mean inter-box Euclidean distance. \\
$k$
  & Number of clusters in the Stage~2 clustering analysis. \\
$d_{kv}$
  & Euclidean distance between the normalised centres of a linked key and value box. \\
$(x_k,y_k), (x_v,y_v)$
  & Normalised centre coordinates of the key box and value box. \\
$\mu_d, \tilde{d}, d_{90}$
  & Mean, median, and 90th percentile of the observed key--value centre distances. \\
$\mathbf{b}_i, w, h$
  & Layout feature vector for token $i$, including its box coordinates, width, and height. \\
$n_h$
  & Number of self-attention heads. \\
$N, d$
  & Number of text tokens and encoder hidden-state dimension. \\
$\mathbf{H}, \mathbf{h}_i$
  & Encoder hidden-state matrix and the hidden state of token $i$. \\
$\mathbf{z}_i, \mathbf{W}, \mathbf{b}$
  & Entity-classification logits, projection matrix, and bias vector for token $i$. \\
$\mathbf{W}_k, \mathbf{W}_v, s_{\text{sem}}(i,j)$
  & Learned key and value projections and their semantic link score. \\
$\Delta x, \Delta y, \theta, s_{i,j}$
  & Centre displacements, their angle, and the final link score for a candidate pair. \\
$n_k, n_v$
  & Numbers of candidate key and value tokens in the V1 linker. \\
$\mathbf{s}_k, T_k$
  & Mean-pooled representation of span $k$ and its set of token indices. \\
$K, V$
  & Numbers of key spans and value spans in the span-level linker. \\
$\mathcal{L}_{\text{entity}}, \mathcal{L}_{\text{link}}, \lambda$
  & Entity loss, link loss, and linker-loss weight. \\
$n_{\text{neg}}, n_{\text{pos}}$
  & Numbers of negative and positive relation candidates in a batch. \\
$p$
  & Dropout probability. \\
$\hat{t}, t$
  & Predicted text and target text in Normalised Edit Distance. \\
$B_p, B_t$
  & Predicted and target bounding boxes in Intersection over Union. \\ \bottomrule
\end{longtable}
\endgroup
